5. Table Normalization (3NF)

First Normal Form (1NF)

All tables satisfy 1NF: every column holds atomic values (no repeating groups), each table has a primary key, and all rows are unique.

Second Normal Form (2NF)

All tables use single-column SERIAL primary keys, so partial dependency cannot exist. Every non-key attribute is fully functionally dependent on the entire PK.

Third Normal Form (3NF)

No transitive dependencies exist. Blood group details (rh_factor, abo_type) live in blood_groups — not repeated in donors/donations. Hospital city lives in hospitals — not repeated in patients. Bank city lives in blood_banks — not in inventory rows. This eliminates update/insertion/deletion anomalies.

Potential AnomalyResolved By
Storing blood group name in every donor rowFK to blood_groups; name stored once
Storing hospital city in every request rowFK to hospitals; city stored once
Storing staff name in every donation rowFK to staff; name stored once
Duplicate inventory entries per (bank, group)UNIQUE(bank_id, blood_group_id, component_type)

6a. DDL — CREATE TABLE (sample: donors)

CREATE TABLE donors (
    donor_id            SERIAL PRIMARY KEY,
    first_name          VARCHAR(50)  NOT NULL,
    last_name           VARCHAR(50)  NOT NULL,
    dob                 DATE         NOT NULL,
    gender              VARCHAR(10)  NOT NULL CHECK(gender IN ('Male','Female','Other')),
    phone               VARCHAR(20)  NOT NULL,
    email               VARCHAR(100),
    address             TEXT,
    city                VARCHAR(50),
    blood_group_id      INTEGER      NOT NULL REFERENCES blood_groups(blood_group_id),
    registration_date   DATE         NOT NULL DEFAULT CURRENT_DATE,
    last_donation_date  DATE,
    total_donations     INTEGER      NOT NULL DEFAULT 0 CHECK(total_donations >= 0),
    weight_kg           NUMERIC(5,1) CHECK(weight_kg IS NULL OR weight_kg >= 50),
    is_active           BOOLEAN      NOT NULL DEFAULT TRUE,
    emergency_contact   TEXT,
    CONSTRAINT chk_age CHECK(dob <= CURRENT_DATE - INTERVAL '18 years')
);

6d. SELECT — WHERE, ORDER BY, GROUP BY

-- WHERE: Find all O- donors in Karachi
SELECT first_name, last_name, phone, total_donations
FROM donors
WHERE city = 'Karachi' AND blood_group_id = 6 AND is_active = TRUE;

-- ORDER BY: Top 5 donors by donation count
SELECT first_name || ' ' || last_name AS name,
       total_donations, last_donation_date
FROM donors
ORDER BY total_donations DESC
LIMIT 5;

-- GROUP BY: Requests per hospital
SELECT h.name AS hospital, COUNT(br.request_id) AS total_requests,
       SUM(br.units_requested) AS total_units_requested
FROM blood_requests br
JOIN hospitals h ON br.hospital_id = h.hospital_id
GROUP BY h.name
ORDER BY total_requests DESC;

-- GROUP BY + HAVING: hospitals with >1 critical request
SELECT h.name, COUNT(*) AS critical_count
FROM blood_requests br
JOIN hospitals h ON br.hospital_id = h.hospital_id
WHERE br.urgency_level = 'critical'
GROUP BY h.name
HAVING COUNT(*) > 1;

6e. Aggregate Functions (SUM, AVG, COUNT, MAX, MIN)

-- Total, average, max donations per blood group
SELECT bg.group_name,
       COUNT(dn.donation_id)            AS total_donations,
       SUM(dn.units_donated)            AS total_units,
       ROUND(AVG(dn.hemoglobin_level),2) AS avg_hemoglobin,
       MAX(dn.units_donated)            AS max_single_donation,
       MIN(dn.hemoglobin_level)         AS min_hemoglobin
FROM donations dn
JOIN blood_groups bg ON dn.blood_group_id = bg.blood_group_id
GROUP BY bg.group_name
ORDER BY total_donations DESC;

-- Total inventory units across all banks
SELECT SUM(units_available)  AS grand_total_units,
       SUM(units_reserved)   AS total_reserved,
       AVG(units_available)  AS avg_per_entry,
       COUNT(DISTINCT bank_id) AS banks_with_stock
FROM blood_inventory
WHERE component_type = 'Whole Blood';

-- Count patients per diagnosis category
SELECT
  CASE WHEN diagnosis ILIKE '%thalassaemia%' THEN 'Thalassaemia'
       WHEN diagnosis ILIKE '%anaemia%'      THEN 'Anaemia'
       WHEN diagnosis ILIKE '%trauma%'       THEN 'Trauma/Emergency'
       ELSE 'Other' END AS category,
  COUNT(*) AS patient_count
FROM patients
GROUP BY category
ORDER BY patient_count DESC;

6f. SQL JOINs

-- INNER JOIN: Donation details (donor + blood group + bank)
SELECT dn.donation_id,
       d.first_name || ' ' || d.last_name AS donor_name,
       bg.group_name, bb.name AS bank_name,
       dn.donation_date::DATE, dn.units_donated, dn.status
FROM donations dn
JOIN donors      d  ON dn.donor_id       = d.donor_id
JOIN blood_groups bg ON dn.blood_group_id = bg.blood_group_id
JOIN blood_banks  bb ON dn.bank_id        = bb.bank_id
ORDER BY dn.donation_date DESC;

-- LEFT JOIN: All patients (even without a request)
SELECT p.first_name || ' ' || p.last_name AS patient,
       h.name AS hospital,
       br.request_id, br.urgency_level, br.status
FROM patients p
LEFT JOIN blood_requests br ON p.patient_id  = br.patient_id
LEFT JOIN hospitals       h ON p.hospital_id = h.hospital_id
ORDER BY p.patient_id;

-- Multi-table JOIN: Full transfusion audit trail
SELECT t.transfusion_id,
       p.first_name || ' ' || p.last_name  AS patient,
       bg.group_name, t.units_transfused,
       t.transfusion_date::DATE,
       s.first_name || ' ' || s.last_name  AS performed_by,
       t.outcome
FROM transfusions t
JOIN blood_requests  br ON t.request_id      = br.request_id
JOIN blood_groups    bg ON br.blood_group_id = bg.blood_group_id
LEFT JOIN patients   p  ON br.patient_id     = p.patient_id
LEFT JOIN staff      s  ON t.staff_id        = s.staff_id
ORDER BY t.transfusion_date DESC;

6g. Complex Queries & CTEs

-- CTE 1: Donors who have donated more than the average
WITH avg_donations AS (
    SELECT AVG(total_donations) AS avg_count FROM donors
)
SELECT d.first_name || ' ' || d.last_name AS donor_name,
       bg.group_name, d.total_donations,
       ROUND(d.total_donations - ad.avg_count, 1) AS above_avg_by
FROM donors d
JOIN blood_groups bg ON d.blood_group_id = bg.blood_group_id
CROSS JOIN avg_donations ad
WHERE d.total_donations > ad.avg_count
ORDER BY d.total_donations DESC;

-- CTE 2: Blood groups with inventory below 50 units (critical supply)
WITH low_stock AS (
    SELECT bg.group_name, bg.blood_group_id,
           SUM(bi.units_available - bi.units_reserved) AS net_units
    FROM blood_inventory bi
    JOIN blood_groups bg ON bi.blood_group_id = bg.blood_group_id
    WHERE bi.component_type = 'Whole Blood'
    GROUP BY bg.group_name, bg.blood_group_id
),
pending_demand AS (
    SELECT blood_group_id, SUM(units_requested) AS total_requested
    FROM blood_requests
    WHERE status IN ('pending','approved')
    GROUP BY blood_group_id
)
SELECT ls.group_name,
       ROUND(ls.net_units,2)          AS available_units,
       COALESCE(pd.total_requested,0) AS pending_units,
       CASE WHEN ls.net_units < COALESCE(pd.total_requested,0)
            THEN 'SHORTAGE' ELSE 'OK' END AS supply_status
FROM low_stock ls
LEFT JOIN pending_demand pd USING (blood_group_id)
ORDER BY ls.net_units ASC;

-- Window Function: Rank donors within each city
SELECT first_name || ' ' || last_name AS donor_name,
       city, total_donations,
       RANK() OVER (PARTITION BY city ORDER BY total_donations DESC) AS city_rank
FROM donors
WHERE is_active = TRUE
ORDER BY city, city_rank;

-- Subquery: Patients at hospitals with most requests
SELECT p.first_name || ' ' || p.last_name AS patient, h.name AS hospital, p.diagnosis
FROM patients p
JOIN hospitals h ON p.hospital_id = h.hospital_id
WHERE h.hospital_id IN (
    SELECT hospital_id
    FROM blood_requests
    GROUP BY hospital_id
    ORDER BY COUNT(*) DESC
    LIMIT 3
);

Supabase Client (backend/supabase.js)

require('dotenv').config();
const { createClient } = require('@supabase/supabase-js');

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_SERVICE_KEY,
  { auth: { persistSession: false } }
);

module.exports = supabase;

Sample API Handler — Parameterized Query (SQL Injection Safe)

// GET /api/inventory?city=Islamabad
app.get('/api/inventory', async (req, res) => {
  const { city, blood_group } = req.query;
  let query = supabase
    .from('v_inventory_detail')
    .select('*')
    .gt('net_available', 0);
  if (city)        query = query.eq('bank_city', city);
  if (blood_group) query = query.eq('group_name', blood_group);
  const { data, error } = await query.order('net_available', { ascending: true });
  if (error) return res.status(500).json({ error: error.message });
  res.json(data);
});

Front-End Fetch API Call (AJAX)

// Fetch donors and render table
async function loadDonors() {
  const res  = await fetch('/api/donors');
  const data = await res.json();
  const tbody = document.querySelector('#donors-table tbody');
  tbody.innerHTML = data.map(d => `
    <tr>
      <td>${d.first_name} ${d.last_name}</td>
      <td>${d.blood_group_name}</td>
      <td>${d.city}</td>
      <td>${d.total_donations}</td>
      <td><button onclick="editDonor(${d.donor_id})">Edit</button></td>
    </tr>`).join('');
}
